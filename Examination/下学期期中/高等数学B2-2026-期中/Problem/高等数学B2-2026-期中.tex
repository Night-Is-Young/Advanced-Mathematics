\begin{center}\Large
    北京大学2025-2026学年第二学期期中试题\\
    高等数学B2
\end{center}
\begin{enumerate}[leftmargin=*,label=\textbf{\arabic*.}]
    \item \textbf{(10\songti{分})}\ 设$D$是平面上的闭圆盘$x^2+y^2\leqslant2x+2y$.计算二重积分
    \[\iint_Dxy\di x\di y.\]

    \item \textbf{(15\songti{分})}\ 设$\Omega$是$\R^3$中球体$x^2+y^2+z^2\leqslant R^2$ $(R>0)$与锥体$z\geqslant\sqrt{x^2+y^2}$的交集.计算三重积分
    \[\iiint_\Omega(x^2+y^2)\di x\di y\di z.\]

    \item \textbf{(15\songti{分})}\ 设$C(R)$是平面上的圆周$x^2+y^2=R^2$ $(R>0)$,定向为逆时针方向.记第二型曲线积分
    \[I(R)=\oint_{C(R)}\dfrac{x\di y-y\di x}{(x^2+xy+y^2)^2}.\]
    证明
    \[\dfrac{13\pi}{9R^2}\leqslant I(R)\leqslant\dfrac{5\pi}{R^2}.\]

    \item \textbf{(15\songti{分})}\ 设$\Sigma$是$\R^3$中圆柱面$x^2+z^2=R^2$ $(R>0)$上满足$x^2+y^2\leqslant R^2$, $z\geqslant0$的部分.计算第一型曲面积分
    \[\iint_\Sigma z^2\di S.\]

    \item \textbf{(15\songti{分})}\ 设$S$是$\R^3$中圆柱面$(x-1)^2+y^2=1$ $(0\leqslant z\leqslant h)$的外侧.计算第二型曲面积分
    \[I=\iint_Sx^2\di y\di z+y^2\di z\di x+z^2\di x\di y.\]

    \item \textbf{(15\songti{分})}\ 求微分方程
    \[y''+4y=\sin x+\cos2x\]
    的通解.

    \item \textbf{(15\songti{分})}\ 求满足方程
    \[y'(x)=xy(x)+x\int_0^1ty(t)\di t\]
    的所有可导函数$y(x)$, $x\in[0,1]$.

\end{enumerate}
