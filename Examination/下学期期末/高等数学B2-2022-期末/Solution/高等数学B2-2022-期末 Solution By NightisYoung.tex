\begin{problem}[1.(10\songti{分})]
求函数
        \[f(x)=\dfrac{\sqrt{|x|}}{2}\ln\dfrac{1+\sqrt{|x|}}{1-\sqrt{|x|}}\]
        在$x=0$处的幂级数展开式,并指出此幂级数的收敛域.
\par\textbf{校订说明：}已核对原件,原题确实含$|x|$.该函数在零点不可导,故原题所要求的普通幂级数展开不存在;解答给出关于$|x|$的收敛展开并说明两侧情形.
\end{problem}
\begin{solution}
令$z=\sqrt{|x|}$.当$|x|<1$时,由
\[\ln\frac{1+z}{1-z}=2\sum_{k=0}^{\infty}\frac{z^{2k+1}}{2k+1}\]
可得
\[f(x)=\sum_{k=0}^{\infty}\frac{|x|^{k+1}}{2k+1},\qquad |x|<1\]
在$|x|=1$时,相应级数$\sum1/(2k+1)$发散,故此关于$|x|$的级数收敛域为$(-1,1)$.
但$f(x)=|x|+O(x^2)$,故$f'_+(0)=1$, $f'_-(0)=-1$,在零点不可导,不存在在零点邻域代表$f$的普通幂级数$\sum c_nx^n$.
在$x\geqslant0$一侧可写为$\sum x^{k+1}/(2k+1)$,在$x\leqslant0$一侧可写为$\sum(-x)^{k+1}/(2k+1)$,两侧不能合并成同一个关于$x$的幂级数.
\end{solution}

\begin{problem}[2.(15\songti{分})]
设$f:\R\to\R$是周期为$2\pi$的函数,$f(x)$在$(-\pi,\pi]$上等于$\e^x$.%
        求$f(x)$的傅里叶级数,以及此傅里叶级数在$x=\pi$处的收敛值.
\end{problem}
\begin{solution}
Fourier系数为
\[a_0=\frac{\e^\pi-\e^{-\pi}}\pi,\quad
a_n=\frac{(-1)^n(\e^\pi-\e^{-\pi})}{\pi(1+n^2)},\quad
b_n=\frac{(-1)^{n+1}n(\e^\pi-\e^{-\pi})}{\pi(1+n^2)}\]
因此Fourier级数为
\[\frac{\e^\pi-\e^{-\pi}}{2\pi}+
\frac{\e^\pi-\e^{-\pi}}\pi\sum_{n=1}^{\infty}\frac{(-1)^n}{1+n^2}
\bigl(\cos(nx)-n\sin(nx)\bigr)\]
由Dirichlet定理,$x=\pi$处级数和为左右极限的平均值,即
\[S(\pi)=\frac{\e^\pi+\e^{-\pi}}2\]
\end{solution}

\begin{problem}[3.(10\songti{分})]
求无穷积分
        \[\int_0^{+\infty}\sqrt{x^3}\e^{-x}\di x\]
        和瑕积分
        \[\int_0^1\sqrt{\dfrac{x^3}{1-x}}\di x\]
        的值.
\end{problem}
\begin{solution}
由Gamma函数及Beta函数公式,
\[\int_0^{+\infty}x^{3/2}\e^{-x}\di x
=\Gamma(5/2)=\frac{3\sqrt\pi}{4}\]
\[\int_0^1x^{3/2}(1-x)^{-1/2}\di x
=\mathrm B(5/2,1/2)=\frac{\Gamma(5/2)\Gamma(1/2)}{\Gamma(3)}=\frac{3\pi}{8}\]
\end{solution}

\begin{problem}[4.(10\songti{分})]
求幂级数
        \[\sum_{n=0}^{\infty}(n+1)(n+2)x^n\]
        的收敛区间,以及此幂级数的和函数.
\end{problem}
\begin{solution}
由系数比值$(n+3)/(n+1)\to1$可知收敛半径为$1$.在$x=\pm1$处通项均不趋于$0$,故收敛域为$(-1,1)$.
当$|x|<1$时,将几何级数逐项求两次导数,得到
\[\sum_{n=0}^{\infty}(n+1)(n+2)x^n
=\frac{\di^2}{\di x^2}\sum_{n=0}^{\infty}x^{n+2}
=\frac{\di^2}{\di x^2}\frac{x^2}{1-x}=\frac2{(1-x)^3}\]
\end{solution}

\begin{problem}[5.(10\songti{分})]
任意给定常数$r>0$,试证明函数项级数
        \[\sum_{n=1}^\infty n^2\e^{-nx}\]
        在$[r,+\infty)$上一致收敛.
\end{problem}
\begin{solution}
对$x\geqslant r$,有$0\leqslant n^2\e^{-nx}\leqslant n^2\e^{-nr}$.而
\[\lim_{n\to\infty}\frac{(n+1)^2\e^{-(n+1)r}}{n^2\e^{-nr}}=\e^{-r}<1\]
故数项级数$\sum n^2\e^{-nr}$收敛.由Weierstrass判别法,题设函数项级数在$[r,+\infty)$上一致收敛.
\end{solution}

\begin{problem}[6.(15\songti{分})]
求函数项级数
        \[\sum_{n=1}^\infty\dfrac{(-1)^n}{n^x+2n}\]
        的收敛域,全体绝对收敛点,全体条件收敛点.
\end{problem}
\begin{solution}
固定$x\in\R$,令$u_n(x)=1/(n^x+2n)$.有$0<u_n(x)\leqslant1/(2n)\to0$.
分母函数$h(y)=y^x+2y$满足$h'(y)=xy^{x-1}+2$.当$x\geqslant0$时导数为正;当$x<0$时$xy^{x-1}\to0$,因此导数最终为正.
故$u_n(x)$最终单调递减,由Leibniz判别法,原级数对所有实数$x$收敛.
当$x>1$时,$u_n(x)\leqslant1/n^x$,绝对值级数收敛.当$x\leqslant1$时,$n^x\leqslant n$,从而
\[u_n(x)\geqslant\frac1{3n}\]
故绝对值级数发散.综上,收敛域为$\R$,绝对收敛点集为$(1,+\infty)$,条件收敛点集为$(-\infty,1]$.
\end{solution}

\begin{problem}[7.(15\songti{分})]
定义函数$\theta:[0,+\infty)\to[0,+\infty)$为
        \[\theta(x)=\int_0^x\sqrt{(t+1)(t+2)(t+3)}\di t\]
        试证明无穷积分
        \[\int_0^{+\infty}\cos\left(\theta(x)\right)\di x\]
        收敛.本题要求写出详细过程和依据.
\end{problem}
\begin{solution}
函数$\theta$连续可导且
\[\theta'(x)=\sqrt{(x+1)(x+2)(x+3)}>0\]
该导数单调递增并趋于$+\infty$,故$\theta$将$[0,+\infty)$严格递增地映到$[0,+\infty)$.
记其反函数为$\phi$.对每个有限$A>0$作换元$u=\theta(x)$,得到
\[\int_0^A\cos(\theta(x))\di x
=\int_0^{\theta(A)}\cos u\,h(u)\di u,\qquad
h(u)=\frac1{\theta'(\phi(u))}\]
由于$\phi$递增,$h(u)$单调递减趋于$0$,而$\left|\int_0^B\cos u\di u\right|\leqslant1$.
由Dirichlet判别法,$\int_0^{+\infty}\cos u\,h(u)\di u$收敛.又$\theta(A)\to+\infty$,因此原无穷积分收敛.该论证适用于任意上限$A\to+\infty$,而非仅适用于选定子序列.
\end{solution}

\begin{problem}[8.(15\songti{分})]
设$n$是正整数.
        \begin{enumerate}[label=\tbf{(\arabic*)},topsep=0pt,parsep=0pt,itemsep=0pt,partopsep=0pt]
            \item \textbf{(5\songti{分})}\ 任意给定$a>0$,试证明含参变量$t$的无穷积分
                \[\int_0^{+\infty}\dfrac{1}{\left(t+x^2\right)^n}\di x\]
                在$[a,+\infty)$上一致收敛.
            \item \textbf{(10\songti{分})}\ 对每个$t\in(0,+\infty)$,求出
                \[\int_0^{+\infty}\dfrac{1}{\left(t+x^2\right)^n}\di x\]
                的值.
        \end{enumerate}
        本题要求写出详细过程和依据.
\end{problem}
\begin{solution}
\begin{enumerate}[label=\tbf{(\arabic*)},topsep=0pt,parsep=0pt,itemsep=0pt,partopsep=0pt]
\item 当$t\geqslant a$时,
\[0<\frac1{(t+x^2)^n}\leqslant
\begin{cases}a^{-n},&0\leqslant x\leqslant1,\\x^{-2},&x\geqslant1.\end{cases}\]
右端可积,故由Weierstrass判别法,该积分在$[a,+\infty)$上一致收敛.
\item 令$I_n(t)$表示所求积分.由第(1)问,在每个$t\in[a,b]\subset(0,+\infty)$上,原积分及偏导数积分$-nI_{n+1}(t)$均一致收敛,因此
\[I_n'(t)=-nI_{n+1}(t),\qquad I_1(t)=\frac\pi{2\sqrt t}\]
迭代求导得
\[I_n(t)=\frac{(-1)^{n-1}}{(n-1)!}I_1^{(n-1)}(t)
=\frac{\pi(2n-3)!!}{2^n(n-1)!\,t^{n-1/2}}\qquad(n\geqslant2)\]
结合$n=1$的情形,也可统一写为
\[I_n(t)=\frac{\sqrt\pi\,\Gamma(n-1/2)}{2\Gamma(n)}t^{1/2-n},\qquad n\geqslant1,\ t>0\]
\end{enumerate}
\end{solution}
