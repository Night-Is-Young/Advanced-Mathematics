\begin{problem}[1.(15\songti{分})]
求函数
        \[f(x)=\dfrac12\ln\dfrac{1+x^2}{1-x^2}\]
        在$x=0$处的幂级数展开式.
\end{problem}
\begin{solution}
当$|x|<1$时,由对数函数的幂级数展开,有
\[\begin{aligned}
f(x)&=\frac12\bigl(\ln(1+x^2)-\ln(1-x^2)\bigr)\\
&=\frac12\left(\sum_{n=1}^{\infty}\frac{(-1)^{n+1}x^{2n}}n+
\sum_{n=1}^{\infty}\frac{x^{2n}}n\right)
=\sum_{k=0}^{\infty}\frac{x^{4k+2}}{2k+1}
\end{aligned}\]
故所求展开式为$\displaystyle\sum_{k=0}^{\infty}x^{4k+2}/(2k+1)$,适用范围为$|x|<1$.
\end{solution}

\begin{problem}[2.(15\songti{分})]
计算下列广义积分的值.
        \begin{enumerate}[label=\tbf{(\arabic*)},topsep=0pt,parsep=0pt,itemsep=0pt,partopsep=0pt]
            \item \textbf{(8\songti{分})}
                \[\int_0^{+\infty}\sqrt{x}\e^{-x}\di x\]
            \item \textbf{(7\songti{分})}\ 
                \[\int_0^1\dfrac{1}{\sqrt{x(1-x)}}\di x\]
        \end{enumerate}
\end{problem}
\begin{solution}
\begin{enumerate}[label=\tbf{(\arabic*)},topsep=0pt,parsep=0pt,itemsep=0pt,partopsep=0pt]
\item $\displaystyle\int_0^{+\infty}\sqrt x\e^{-x}\di x
=\Gamma(3/2)=\frac12\Gamma(1/2)=\frac{\sqrt\pi}{2}$.
\item $\displaystyle\int_0^1\frac{\di x}{\sqrt{x(1-x)}}
=\mathrm B(1/2,1/2)=\frac{\Gamma(1/2)^2}{\Gamma(1)}=\pi$.
\end{enumerate}
\end{solution}

\begin{problem}[3.(15\songti{分})]
求幂级数
        \[\sum_{n=1}^{\infty}(n+1)x^n\]
        的收敛区间及其和函数.
\end{problem}
\begin{solution}
由系数比值$\displaystyle\frac{n+2}{n+1}\to1$可知收敛半径为$1$.当$x=\pm1$时通项均不趋于$0$,所以收敛区间及收敛域均为$(-1,1)$.
在$|x|<1$时,对几何级数逐项求导得
\[S(x)=\frac{\di}{\di x}\sum_{n=1}^{\infty}x^{n+1}
=\frac{\di}{\di x}\frac{x^2}{1-x}=\frac{x(2-x)}{(1-x)^2}\]
\end{solution}

\begin{problem}[4.(15\songti{分})]
任意取定$r>0$,证明含参变量$y$的无穷积分
        \[\int_0^{+\infty}\e^{-xy^2}\cos x\di x\]
        在$y\in[r,+\infty)$上一致收敛.
\end{problem}
\begin{solution}
对任意$y\geqslant r$和$x\geqslant0$,有
\[|\e^{-xy^2}\cos x|\leqslant\e^{-r^2x}\]
右端在$[0,+\infty)$上可积.更具体地,
\[\sup_{y\geqslant r}\int_A^{+\infty}|\e^{-xy^2}\cos x|\di x
\leqslant\frac{\e^{-r^2A}}{r^2}\to0\qquad(A\to+\infty)\]
故原积分在$y\in[r,+\infty)$上一致收敛,而且绝对值积分也一致收敛.
\end{solution}

\begin{problem}[5.(10\songti{分})]
求函数项级数
        \[\sum_{n=1}^\infty\dfrac{(-1)^n}{n^x+n}\]
        的收敛域.
\end{problem}
\begin{solution}
固定$x\in\R$,令$u_n(x)=1/(n^x+n)$.有$0<u_n(x)\leqslant1/n\to0$.
考虑分母函数$h(y)=y^x+y$,其导数为$h'(y)=xy^{x-1}+1$.当$x\geqslant0$时$h'(y)>0$;当$x<0$时$xy^{x-1}\to0$,故$y$充分大时仍有$h'(y)>0$.
所以对每个固定$x$,数列$u_n(x)$最终单调递减趋于$0$.由Leibniz判别法,原级数对每个$x\in\R$都收敛,收敛域为$\R$.
\end{solution}

\begin{problem}[6.(20\songti{分})]
回答下列问题.
        \begin{enumerate}[label=\tbf{(\arabic*)},topsep=0pt,parsep=0pt,itemsep=0pt,partopsep=0pt]
            \item \textbf{(10\songti{分})}\ 设$p\in\R$且不是整数,定义在$(-\infty,+\infty)$上的函数$f(x)$以$2\pi$为周期,它在$(-\pi,\pi)$上等于$\cos(px)$.%
                求出$f(x)$的傅里叶级数及其和函数.
            \item \textbf{(3\songti{分})}\ 根据\tbf{(1)}的结论证明:当$t\in\R$且$\dfrac{t}{\pi}$不是整数时,有
                \[\dfrac{1}{\sin t}=\dfrac1t+\sum_{n=1}^{\infty}(-1)^n\left(\dfrac{1}{t+n\pi}+\dfrac{1}{t-n\pi}\right)\]
            \item \textbf{(7\songti{分})}\ 根据\tbf{(2)}的结论证明
                \[\int_0^{+\infty}\dfrac{\sin t}{t}\di t=\dfrac\pi2\]

        \end{enumerate}
\end{problem}
\begin{solution}
\begin{enumerate}[label=\tbf{(\arabic*)},topsep=0pt,parsep=0pt,itemsep=0pt,partopsep=0pt]
\item 由于$\cos(px)$为偶函数,有$b_n=0$,且
\[a_0=\frac{2\sin(p\pi)}{p\pi},\qquad
a_n=\frac{(-1)^n\sin(p\pi)}\pi\left(\frac1{p+n}+\frac1{p-n}\right)\]
因此Fourier级数为
\[\frac{\sin(p\pi)}{p\pi}+\frac{\sin(p\pi)}\pi
\sum_{n=1}^{\infty}(-1)^n\left(\frac1{p+n}+\frac1{p-n}\right)\cos(nx)\]
周期延拓在一个周期内分段光滑,两端的单侧极限都为$\cos(p\pi)$.由Dirichlet定理,和函数为
\[S(x)=\cos\bigl(p(x-2k\pi)\bigr),\quad
x\in[(2k-1)\pi,(2k+1)\pi],\ k\in\mathbb Z\]
特别地,在$[-\pi,\pi]$上$S(x)=\cos(px)$;不能将此式不加周期延拓地用于整个$\R$.
\item 在上一展开式中取$x=0$,再令$p=t/\pi$.由于$\sin t\neq0$,两边除以$\sin t$,即得
\[\frac1{\sin t}=\frac1t+\sum_{n=1}^{\infty}(-1)^n
\left(\frac1{t+n\pi}+\frac1{t-n\pi}\right)\]
\item 首先由Dirichlet判别法,$\int_0^{+\infty}\sin t/t\di t$收敛,其中在$t=0$处按连续延拓理解.
对$0\leqslant x\leqslant\pi/2$,令
\[A_n=\int_0^{\pi/2}\frac{\sin x}{x+n\pi}\di x,\qquad
B_n=\int_0^{\pi/2}\frac{\sin x}{x-n\pi}\di x\quad(n\geqslant1)\]
并令$A_0=\int_0^{\pi/2}\sin x/x\di x$.把$[0,(N+1)\pi]$分成半周期区间,分别平移和反射,可得有限和恒等式
\[\int_0^{(N+1)\pi}\frac{\sin t}{t}\di t
=A_0+\sum_{n=1}^{N}(-1)^n(A_n+B_n)+(-1)^{N+1}B_{N+1}\]
其中$|B_{N+1}|\leqslant\dfrac{\pi/2}{(N+1)\pi-\pi/2}\to0$.
又对$n\geqslant1$和$0\leqslant x\leqslant\pi/2$有
\[\left|\frac{\sin x}{x+n\pi}+\frac{\sin x}{x-n\pi}\right|
=\frac{2x\sin x}{n^2\pi^2-x^2}
\leqslant\frac1{\pi(n^2-1/4)}\]
右端所成级数收敛,故可一致地逐项积分.由第(2)问,在$0<x\leqslant\pi/2$时括号内级数等于$1-\sin x/x$,此等式在$x=0$处也连续成立.于是
\[\begin{aligned}
\int_0^{+\infty}\frac{\sin t}{t}\di t
&=\int_0^{\pi/2}\frac{\sin x}{x}\di x+
\int_0^{\pi/2}\left(1-\frac{\sin x}{x}\right)\di x=\frac\pi2
\end{aligned}\]
\end{enumerate}
\end{solution}

\begin{problem}[7.(10\songti{分})]
设$f:(0,+\infty)\to\R$是单调递减的连续函数(没有假定其导函数$f'(x)$的存在).%
        $C,D\in\R$满足
        \[\lim_{x\to0^+}f(x)=C\ \ \ \ \ \lim_{x\to+\infty}f(x)=D\]
        对于$0<a<b$,求广义积分
        \[\int_0^{+\infty}\dfrac{f(ax)-f(bx)}{x}\di x\]
        的值.
\end{problem}
\begin{solution}
对$0<\delta<A$,分别换元并抵消公共区间,得到
\[\int_\delta^A\frac{f(ax)-f(bx)}x\di x
=\int_{a\delta}^{b\delta}\frac{f(u)}u\di u-
\int_{aA}^{bA}\frac{f(u)}u\di u\]
由$f$单调递减及$\ln(b/a)>0$,有
\[f(b\delta)\ln\frac ba\leqslant
\int_{a\delta}^{b\delta}\frac{f(u)}u\di u
\leqslant f(a\delta)\ln\frac ba\]
令$\delta\to0^+$,中间积分趋于$C\ln(b/a)$.同理,
\[f(bA)\ln\frac ba\leqslant\int_{aA}^{bA}\frac{f(u)}u\di u
\leqslant f(aA)\ln\frac ba\]
令$A\to+\infty$,中间积分趋于$D\ln(b/a)$.由于被积函数非负,且上述两个端点极限均有限,原广义积分收敛,其值为
\[\int_0^{+\infty}\frac{f(ax)-f(bx)}x\di x=(C-D)\ln\frac ba\]
\end{solution}
