\begin{problem}[1.(10\songti{分})]
求幂级数
        \[\sum_{n=1}^{\infty}\dfrac{x^{n^3}}{10^n}\]
        的收敛域.
\end{problem}
\begin{solution}
令$u_n(x)=x^{n^3}/10^n$,则
\[\sqrt[n]{|u_n(x)|}=\frac{|x|^{n^2}}{10}\to
\begin{cases}0,&|x|<1,\\1/10,&|x|=1,\\+\infty,&|x|>1.\end{cases}\]
由根值判别法,原级数在$|x|\leqslant1$时绝对收敛,在$|x|>1$时发散,故收敛域为$[-1,1]$.
其中$x=1$时级数和为$1/9$,而$x=-1$时级数和为$\sum_{n=1}^{\infty}(-1)^n/10^n=-1/11$,不能将两端点的级数都写成正项几何级数.
\end{solution}

\begin{problem}[2.(10\songti{分})]
在$(-1,1)$上将函数
        \[f(x)=\arctan x+\dfrac12\ln\dfrac{1+x}{1-x}\]
        展开为幂级数.
\end{problem}
\begin{solution}
当$|x|<1$时,有
\[\arctan x=\sum_{n=0}^{\infty}\frac{(-1)^nx^{2n+1}}{2n+1},\qquad
\frac12\ln\frac{1+x}{1-x}=\sum_{n=0}^{\infty}\frac{x^{2n+1}}{2n+1}\]
相加后仅保留$n$为偶数的项,得到
\[f(x)=2\sum_{k=0}^{\infty}\frac{x^{4k+1}}{4k+1},\qquad -1<x<1\]
\end{solution}

\begin{problem}[3.(10\songti{分})]
求瑕积分
        \[\int_0^1\sqrt{\dfrac{x^5}{1-x}}\dx\]
        的值.
\end{problem}
\begin{solution}
由Beta函数公式,
\[\int_0^1\sqrt{\frac{x^5}{1-x}}\di x
=\mathrm B(7/2,1/2)=\frac{\Gamma(7/2)\Gamma(1/2)}{\Gamma(4)}=\frac{5\pi}{16}\]
\end{solution}

\begin{problem}[4.(10\songti{分})]
判断级数
        \[\sum_{n=1}^{\infty}\dfrac{\sin(2n)}{n+\dfrac1n}\left(1+\dfrac1n\right)^n\]
        的敛散性.
\end{problem}
\begin{solution}
令$b_n=\dfrac{n}{n^2+1}(1+1/n)^n$.对实变量$x>0$,设
\[g(x)=\ln x-\ln(x^2+1)+x\ln(1+1/x)\]
由$\ln(1+1/x)<1/x$可得
\[g'(x)<\frac2x-\frac{2x}{x^2+1}-\frac1{x+1}
=\frac{-x^3+x+2}{x(x+1)(x^2+1)}<0\qquad(x\geqslant2)\]
因此$b_n$最终单调递减,且$b_n\to0$.另一方面,
\[\sum_{n=1}^N\sin(2n)=\frac{\sin N\sin(N+1)}{\sin1}\]
部分和有界,由Dirichlet判别法可知原级数收敛.
又$nb_n\to\e>1$,故对某个$N_0$及所有$n\geqslant N_0$有$b_n\geqslant1/n$.于是
\[\sum_{n=N_0}^N b_n|\sin(2n)|\geqslant
\sum_{n=N_0}^N\frac{\sin^2(2n)}n
=\frac12\sum_{n=N_0}^N\frac1n-\frac12\sum_{n=N_0}^N\frac{\cos(4n)}n\]
后一余弦级数由Dirichlet判别法收敛,前一调和级数发散,故绝对值级数发散.结论为条件收敛.
\end{solution}

\begin{problem}[5.(10\songti{分})]
设$E\in\R$.
        \begin{enumerate}[label=\tbf{(\arabic*)},topsep=0pt,parsep=0pt,itemsep=0pt,partopsep=0pt]
            \item \textbf{(5\songti{分})}\ 求出所有$E\in\R$使得
                \[\int_0^{+\infty}\sum_{n=0}^{\infty}\left(\dfrac{(Ex)^n}{n!}\e^{-x}\right)\di x\]
                收敛.
            \item \textbf{(5\songti{分})}\ 求出所有$E\in\R$使得
                \[\sum_{n=0}^{\infty}\int_0^{+\infty}\left(\dfrac{(Ex)^n}{n!}\e^{-x}\right)\di x\]
                收敛.本小问的结果可以用$\Gamma$函数表示.
        \end{enumerate}
\end{problem}
\begin{solution}
\begin{enumerate}[label=\tbf{(\arabic*)},topsep=0pt,parsep=0pt,itemsep=0pt,partopsep=0pt]
\item 对每个$E\in\R$和$x\geqslant0$,先求和得$\sum_{n=0}^{\infty}(Ex)^n/n!=\e^{Ex}$.故所求积分化为
\[\int_0^{+\infty}\e^{(E-1)x}\di x\]
当且仅当$E<1$时收敛,其值为$1/(1-E)$;当$E\geqslant1$时发散.
\item 先积分得
\[\int_0^{+\infty}\frac{(Ex)^n}{n!}\e^{-x}\di x
=\frac{E^n\Gamma(n+1)}{n!}=E^n\]
故所求级数是$\sum_{n=0}^{\infty}E^n$,当且仅当$|E|<1$时收敛,其和为$1/(1-E)$.在$|E|\geqslant1$时通项不趋于零.
\end{enumerate}
\end{solution}

\begin{problem}[6.(10\songti{分})]
对于每个$x\in[0,1],n=1,2,\cdots$,定义
        \[f_1(x)=\int_0^x\sqrt{1+t^4}\di t\ \ \ \ \ f_{n+1}(x)=\int_0^xf_n(t)\di t\]
        试证明
        \[\sum_{n=1}^{\infty}f_n(x)\]
        在$[0,1]$上一致收敛.
\end{problem}
\begin{solution}
对$0\leqslant t\leqslant1$,有$\sqrt{1+t^4}\leqslant\sqrt2$,故$0\leqslant f_1(x)\leqslant\sqrt2x$.
由归纳法,若$0\leqslant f_n(x)\leqslant\sqrt2x^n/n!$,则
\[0\leqslant f_{n+1}(x)\leqslant\int_0^x\frac{\sqrt2t^n}{n!}\di t
=\frac{\sqrt2x^{n+1}}{(n+1)!}\]
因此对所有$n\geqslant1$, $x\in[0,1]$,有$|f_n(x)|\leqslant\sqrt2/n!$.
由于$\sum\sqrt2/n!$收敛,由Weierstrass判别法,原级数在$[0,1]$上一致收敛.
\end{solution}

\begin{problem}[7.(15\songti{分})]
设$b\in\R$.
        \begin{enumerate}[label=\tbf{(\arabic*)},topsep=0pt,parsep=0pt,itemsep=0pt,partopsep=0pt]
            \item \textbf{(5\songti{分})}\ 试证明含参变量$b$的无穷积分
                \[\int_0^{+\infty}x\e^{-x^2}\cos(2bx)\di x\]
                在$(-\infty,+\infty)$上一致收敛.
            \item \textbf{(10\songti{分})}\ 试证明
                \[\int_0^{+\infty}\e^{-x^2}\sin(2bx)\di x=\e^{-b^2}\int_0^b\e^{t^2}\di t\]
        \end{enumerate}
\end{problem}
\begin{solution}
\begin{enumerate}[label=\tbf{(\arabic*)},topsep=0pt,parsep=0pt,itemsep=0pt,partopsep=0pt]
\item 对所有$b\in\R$,有$|x\e^{-x^2}\cos(2bx)|\leqslant x\e^{-x^2}$,且
\[\int_0^{+\infty}x\e^{-x^2}\di x=\frac12\]
由Weierstrass判别法,题设积分在$\R$上一致收敛.
\item 记$J(b)=\int_0^{+\infty}\e^{-x^2}\sin(2bx)\di x$.原被积函数及其对$b$的偏导数分别由可积函数$\e^{-x^2}$和$2x\e^{-x^2}$控制,因此可求导,且
\[J'(b)=2I(b),\qquad I(b)=\int_0^{+\infty}x\e^{-x^2}\cos(2bx)\di x\]
分部积分得
\[I(b)=-\frac12\left.\e^{-x^2}\cos(2bx)\right|_0^{+\infty}
-b\int_0^{+\infty}\e^{-x^2}\sin(2bx)\di x=\frac12-bJ(b)\]
因此$J'(b)+2bJ(b)=1$.乘积分因子$\e^{b^2}$并利用$J(0)=0$,得到
\[\bigl(\e^{b^2}J(b)\bigr)'=\e^{b^2},\qquad
J(b)=\e^{-b^2}\int_0^b\e^{t^2}\di t\]
对正、负$b$均成立.
\end{enumerate}
\end{solution}

\begin{problem}[8.(15\songti{分})]
\begin{enumerate}[label=\tbf{(\arabic*)},topsep=0pt,parsep=0pt,itemsep=0pt,partopsep=0pt]
            \item \textbf{(10\songti{分})}\ 设$f:\R\to\R$是周期为$2\pi$的函数,$\forall x\in(-\pi,\pi),f(x)=\e^x$.%
                求出$f(x)$的傅里叶级数,并求出$f(x)$的傅里叶级数在$x=\pi$处的和.
            \item \textbf{(5\songti{分})}\ 求出级数
                \[\sum_{n=1}^{\infty}\dfrac{1}{1+n^2}\]
                的和.
        \end{enumerate}
\end{problem}
\begin{solution}
\begin{enumerate}[label=\tbf{(\arabic*)},topsep=0pt,parsep=0pt,itemsep=0pt,partopsep=0pt]
\item 直接积分得
\[a_0=\frac{\e^\pi-\e^{-\pi}}\pi,\quad
a_n=\frac{(-1)^n(\e^\pi-\e^{-\pi})}{\pi(1+n^2)},\quad
b_n=\frac{(-1)^{n+1}n(\e^\pi-\e^{-\pi})}{\pi(1+n^2)}\]
故Fourier级数为
\[\frac{\e^\pi-\e^{-\pi}}{2\pi}+
\frac{\e^\pi-\e^{-\pi}}\pi\sum_{n=1}^{\infty}\frac{(-1)^n}{1+n^2}
\bigl(\cos(nx)-n\sin(nx)\bigr)\]
由Dirichlet定理,$\displaystyle S(\pi)=(\e^\pi+\e^{-\pi})/2$.
\item 在上述展开式中令$x=\pi$,得到
\[\frac{\e^\pi+\e^{-\pi}}2
=\frac{\e^\pi-\e^{-\pi}}{2\pi}+
\frac{\e^\pi-\e^{-\pi}}\pi\sum_{n=1}^{\infty}\frac1{1+n^2}\]
解得
\[\sum_{n=1}^{\infty}\frac1{1+n^2}
=\frac{\pi(\e^\pi+\e^{-\pi})}{2(\e^\pi-\e^{-\pi})}-\frac12\]
\end{enumerate}
\end{solution}

\begin{problem}[9.(10\songti{分})]
设级数
        \[\sum_{n=0}^{\infty}a_n\]
        收敛,每项$a_n>0$.$T$是序列$\left\{a_n\right\}$中的最大项.对于任意$x\in\R$,定义$L(x)$是序列$\left\{a_n\right\}$中大于$x$的项的个数.
        \begin{enumerate}[label=\tbf{(\arabic*)},topsep=0pt,parsep=0pt,itemsep=0pt,partopsep=0pt]
            \item \textbf{(2\songti{分})}\ 试证明$0$是$L(x)$的瑕点.
            \item \textbf{(8\songti{分})}\ 试证明瑕积分
                \[\int_0^TL(x)\di x\]
                收敛,并且
                \[\int_0^TL(x)\di x=\sum_{n=0}^{\infty}a_n\]
        \end{enumerate}
\end{problem}
\begin{solution}
\begin{enumerate}[label=\tbf{(\arabic*)},topsep=0pt,parsep=0pt,itemsep=0pt,partopsep=0pt]
\item 由级数收敛可知$a_n\to0$,故对每个$x>0$,大于$x$的项仅有有限多个,即$L(x)<+\infty$.
任取正整数$N$,令$\delta=\min\{a_0,\ldots,a_{N-1}\}>0$,则$0<x<\delta$时$L(x)\geqslant N$.因此$L(x)\to+\infty$ $(x\to0^+)$,零点是瑕点.
\item 对每个$\varepsilon>0$,在$[\varepsilon,T]$上只有有限多个$a_n$可产生跳跃,故$L$可积.逐项计算这个有限和可得
\[\int_\varepsilon^T L(x)\di x
=\sum_{a_n>\varepsilon}(a_n-\varepsilon)\leqslant\sum_{n=0}^{\infty}a_n\]
被积函数非负,故当$\varepsilon\to0^+$时左端单调递增且有上界,瑕积分收敛,记其值为$I$.
另一方面,对任意固定$N$,有
\[\int_\varepsilon^T L(x)\di x\geqslant
\sum_{n=0}^N\max\{a_n-\varepsilon,0\}\]
先令$\varepsilon\to0^+$,再令$N\to\infty$,得到$I\geqslant\sum_{n=0}^{\infty}a_n$.结合上界即得
\[\int_0^T L(x)\di x=\sum_{n=0}^{\infty}a_n\]
\end{enumerate}
\end{solution}
