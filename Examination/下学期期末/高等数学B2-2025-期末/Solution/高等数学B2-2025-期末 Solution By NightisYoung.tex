\begin{problem}[1.(14\songti{分})]
    讨论下列级数的敛散性.
    \begin{enumerate}[label=\tbf{(\arabic*)},topsep=0pt,parsep=0pt,itemsep=0pt,partopsep=0pt]
        \item \[\sum_{n=1}^{\infty}2^n\left(\dfrac{n-1}{n}\right)^{n^2}\]
        \item \[\sum_{n=2}^{\infty}\dfrac{1}{n\sqrt{\ln n}}\]
    \end{enumerate}

\end{problem}
\begin{solution}
    \begin{enumerate}[label=\tbf{(\arabic*)},topsep=0pt,parsep=0pt,itemsep=0pt,partopsep=0pt]
        \item 令$u_n=2^n\left(\dfrac{n-1}{n}\right)^{n^2}$,于是
            \[\lim_{n\to\infty}\sqrt[n]{u_n}=2\lim_{n\to\infty}\left(1-\dfrac1n\right)^n=\dfrac{2}{\e}<1\]
            根据Cauchy判别法可知原级数收敛.
        \item 令$u_n=\dfrac{1}{n\sqrt{\ln n}}$.函数$\dfrac{1}{x\sqrt{\ln x}}$在$[2,+\infty)$上为正且单调递减,于是
            \[\sum_{n=2}^{N}u_n\geqslant\int_2^{N+1}\dfrac{1}{x\sqrt{\ln x}}\di x
            =2\left(\sqrt{\ln(N+1)}-\sqrt{\ln2}\right)\]
            右端在$N\to\infty$时趋于$+\infty$,因而原级数发散于$+\infty$.
    \end{enumerate}
\end{solution}
\begin{problem}[2.(14\songti{分})]
    判断下列级数的敛散性.如果收敛,请判断其为绝对收敛还是条件收敛.
    \begin{enumerate}[label=\tbf{(\arabic*)},topsep=0pt,parsep=0pt,itemsep=0pt,partopsep=0pt]
        \item \[\sum_{n=2}^{\infty}\dfrac{(-1)^n}{\sqrt{n}+(-1)^n}\]
        \item \[\sum_{n=1}^{\infty}\dfrac{(-1)^{n-1}}{n^{1+\frac1n}}\]
    \end{enumerate}

\end{problem}
\begin{solution}
    \begin{enumerate}[label=\tbf{(\arabic*)},topsep=0pt,parsep=0pt,itemsep=0pt,partopsep=0pt]
        \item 我们有
            \[\dfrac{(-1)^n}{\sqrt{n}+(-1)^n}=\dfrac{(-1)^n\sqrt{n}-1}{n-1}\]
            于是
            \[\sum_{n=2}^{N}\dfrac{(-1)^n}{\sqrt{n}+(-1)^n}
            =\sum_{n=2}^{N}\dfrac{\sqrt{n}}{n-1}(-1)^{n}-\sum_{n=2}^{N}\dfrac{1}{n-1}\]
            注意到$a_n=\dfrac{\sqrt{n}}{n-1}$对$n$单调递减且有
            \[\lim_{n\to\infty}a_n=\lim_{n\to\infty}\dfrac{1}{\sqrt{n}-\dfrac{1}{\sqrt{n}}}=0\]
            于是根据Leibniz判别法可知第一部分收敛.又因为级数
            \[\sum_{n=2}^{\infty}\dfrac{1}{n-1}=\sum_{n=1}^{\infty}\dfrac1n\]
            发散于$+\infty$,从而原级数发散于$-\infty$.
        \item 令$u_n=\dfrac{1}{n^{1+\frac1n}}$,于是$\ln u_n=-\dfrac{n+1}{n}\ln n$.令$f(x)=-\dfrac{x+1}{x}\ln x$,则有
            \[f'(x)=\dfrac{\ln x}{x^2}-\dfrac{x+1}{x^2}=\dfrac{\ln x-x-1}{x^2}\]
            当$x\geqslant1$时,由$\ln x\leqslant x-1$可知$f'(x)<0$,即$f(x)$在$[1,+\infty)$上单调递减.于是$\left\{u_n\right\}_{n=1}^{\infty}$单调递减.此外有
            \[\lim_{n\to\infty}u_n=\lim_{n\to\infty}\dfrac{1}{n}\cdot\dfrac{1}{n^{\frac1n}}=0\cdot1=0\]
            于是根据Leibniz判别法可知原级数收敛.\\
            现在考察其绝对收敛性.注意到
            \[\lim_{n\to\infty}\dfrac{u_n}{\frac1n}=\lim_{n\to\infty}\dfrac{1}{n^{\frac1n}}=1\]
            又因为级数
            \[\sum_{n=1}^{\infty}\dfrac1n\]
            发散,于是根据比较判别法可知
            \[\sum_{n=1}^{\infty}\left|\dfrac{(-1)^{n-1}}{n^{1+\frac1n}}\right|=\sum_{n=1}^{\infty}u_n\]
            发散.于是原级数条件收敛.
    \end{enumerate}
\end{solution}
\begin{problem}[3.(16\songti{分})]
    求幂级数
    \[\sum_{n=1}^{\infty}\dfrac{(-1)^{n-1}}{(2n-1)(2n+1)}x^{2n+1}\]
    的收敛半径,收敛区间,收敛域以及和函数.

\end{problem}
\begin{solution}
    令$u_n(x)=\dfrac{(-1)^{n-1}}{(2n-1)(2n+1)}x^{2n+1}$.当$x\neq0$时有
    \[\lim_{n\to\infty}\left|\dfrac{u_{n+1}(x)}{u_n(x)}\right|
    =\lim_{n\to\infty}\dfrac{2n-1}{2n+3}x^2=x^2\]
    于是原级数在$|x|<1$时绝对收敛,在$|x|>1$时发散.因此收敛半径$R=1$,收敛区间为$(-1,1)$.又当$|x|=1$时有
    \[\sum_{n=1}^{\infty}|u_n(x)|
    =\sum_{n=1}^{\infty}\dfrac{1}{(2n-1)(2n+1)}
    =\dfrac12\lim_{N\to\infty}\left(1-\dfrac{1}{2N+1}\right)=\dfrac12\]
    即两个端点处均绝对收敛,因此该幂级数的收敛域为$[-1,1]$.令和函数为$S(x)$,当$|x|<1$时逐项求导有
    \[S'(x)=\sum_{n=1}^{\infty}\dfrac{(-1)^{n-1}}{2n-1}x^{2n}=x\sum_{n=1}^{\infty}\dfrac{(-1)^{n-1}}{2n-1}x^{2n-1}=x\arctan x\]
    于是
    \[\begin{aligned}
        S(x)-S(0)
        &= \int_0^x t\arctan t\di t \\
        &= \dfrac{x^2}{2}\arctan x-\dfrac12\int_0^x\dfrac{t^2}{1+t^2}\di t \\
        &= \dfrac12\left((1+x^2)\arctan x-x\right)
    \end{aligned}\]
    又因为$S(0)=0$,于是$|x|<1$时$S(x)=\dfrac12\left((1+x^2)\arctan x-x\right)$.
    对所有$x\in[-1,1]$,有$|u_n(x)|\leqslant\dfrac{1}{(2n-1)(2n+1)}$,由Weierstrass判别法可知原级数在$[-1,1]$上一致收敛,从而$S(x)$在$[-1,1]$上连续.于是端点处也适用上述公式,即
    \[S(x)=\dfrac12\left((1+x^2)\arctan x-x\right),\qquad x\in[-1,1]\]
    其中$S(1)=\dfrac\pi4-\dfrac12$, $S(-1)=-\dfrac\pi4+\dfrac12$.

\end{solution}
\begin{problem}[4.(12\songti{分})]
    求含参变量$a$的积分
    \[I(a)=\int_0^{\frac\pi2}\ln\left(a^2\sin^2x+\cos^2x\right)\di x\qquad(a>0)\]
        
\end{problem}
\begin{solution}
    记$f(x,a)=\ln\left(a^2\sin^2x+\cos^2x\right)$.对于任意$0<a_0<a_1$,函数$f(x,a)$及其偏导数
    \[\dfrac{\p f}{\p a}(x,a)=\dfrac{2a\sin^2x}{a^2\sin^2x+\cos^2x}\]
    都在$\left[0,\dfrac\pi2\right]\times[a_0,a_1]$上连续,因此对任意$a>0$都可以在积分号下求导,得到
    \[I'(a)
    =\int_0^{\frac\pi2}\dfrac{\p f}{\p a}(x,a)\di x
    =\int_0^{\frac\pi2}\dfrac{2a\sin^2x}{a^2\sin^2x+\cos^2x}\di x
    \xlongequal{u=\tan x}\int_0^{+\infty}\dfrac{2au^2}{\left(a^2u^2+1\right)\left(u^2+1\right)}\di u\]
    当$a=1$时有
    \[I'(1)=\int_0^{+\infty}\dfrac{2u^2}{\left(1+u^2\right)^2}\di u=\left.\left(\arctan u-\dfrac{u}{1+u^2}\right)\right|_0^{+\infty}=\dfrac\pi2\]
    当$a\neq1$时有
    \[\begin{aligned}
        I'(a)
        &= \int_0^{+\infty}\dfrac{2au^2}{\left(a^2u^2+1\right)\left(u^2+1\right)}\di u \\
        &= \dfrac{2a}{1-a^2}\int_0^{+\infty}\left(\dfrac{1}{a^2u^2+1}-\dfrac{1}{u^2+1}\right)\di u \\
        &= \left.\dfrac{2a}{1-a^2}\left(\dfrac1a\arctan(au)-\arctan u\right)\right|_0^{+\infty} \\
        &= \dfrac{2a}{1-a^2}\left(\dfrac1a-1\right)\dfrac\pi2 \\
        &= \dfrac{\pi}{1+a}
    \end{aligned}\]
    结合$a=1$的情形,对所有$a>0$均有$I'(a)=\dfrac{\pi}{1+a}$.从而$I(a)=\pi\ln(1+a)+C$.又因为
    \[I(1)=\int_0^{\frac\pi2}\ln\left(\sin^2x+\cos^2x\right)\di x=\int_0^{\frac\pi2}0\di x=0\]
    于是$a>0$时
    \[I(a)=\pi\ln\dfrac{1+a}{2}\qquad(a>0)\]

\end{solution}
\begin{problem}[5.(12\songti{分})]
    判断广义积分
    \[\int_0^{+\infty}\dfrac{1}{\left(1+x^2\right)\left(\sin^2 x\right)^\alpha}\di x\]
    的敛散性,其中$0<\alpha<\dfrac12$.
\end{problem}
\begin{solution}
    先考虑瑕积分
    \[\int_0^{\frac\pi2}\dfrac{1}{\left(\sin x\right)^{2\alpha}}\di x\]
    瑕点为$0$.由于
    \[\lim_{x\to0^+}\dfrac{\frac{1}{\left(\sin x\right)^{2\alpha}}}{\frac{1}{x^{2\alpha}}}=\lim_{x\to0^+}\left(\dfrac{x}{\sin x}\right)^{2\alpha}=1\]
    于是上述瑕积分与
    \[\int_0^{\frac\pi2}\dfrac{1}{x^{2\alpha}}\di x\]
    同敛散.又因为$0<2\alpha<1$,从而上述瑕积分收敛,因而开始的瑕积分亦收敛.记其值为$C$.由对称性可知
    \[\int_0^\pi\dfrac{1}{(\sin^2u)^\alpha}\di u=2C<+\infty\]
    对每个$n=0,1,2,\ldots$,令
    \[\begin{aligned}
        B_n
        &=\int_{n\pi}^{(n+1)\pi}\dfrac{1}{(1+x^2)(\sin^2x)^\alpha}\di x \\
        &\xlongequal{u=x-n\pi}\int_0^\pi\dfrac{1}{(1+(u+n\pi)^2)(\sin^2u)^\alpha}\di u \\
        &\leqslant\dfrac{1}{1+n^2\pi^2}\int_0^\pi\dfrac{1}{(\sin^2u)^\alpha}\di u
        =\dfrac{2C}{1+n^2\pi^2}
    \end{aligned}\]
    因此各段瑕积分均收敛,且由比较判别法可知$\displaystyle\sum_{n=0}^{\infty}B_n$收敛.
    对任意$A\geqslant\pi$,取整数$m\geqslant1$,使$m\pi\leqslant A<(m+1)\pi$,则
    \[\int_0^A\dfrac{1}{(1+x^2)(\sin^2x)^\alpha}\di x
    =\sum_{n=0}^{m-1}B_n+R_A,\qquad
    R_A=\int_{m\pi}^A\dfrac{1}{(1+x^2)(\sin^2x)^\alpha}\di x\]
    当$A=m\pi$时约定$R_A=0$.由于
    \[0\leqslant R_A\leqslant B_m\leqslant\dfrac{2C}{1+m^2\pi^2}\to0\qquad(A\to+\infty)\]
    于是原广义积分收敛,并且
    \[\int_0^{+\infty}\dfrac{1}{(1+x^2)(\sin^2x)^\alpha}\di x
    =\sum_{n=0}^{\infty}B_n
    \leqslant2C\sum_{n=0}^{\infty}\dfrac{1}{1+n^2\pi^2}<+\infty\]
\end{solution}
\begin{problem}[6.(12\songti{分})]
    讨论积分
    \[\int_1^{+\infty}t\e^{-tx}\dfrac{\cos x}{x}\di x\]
    在$0\leqslant t<+\infty$上的一致收敛性.
\end{problem}
\begin{solution}
    对于$y\geqslant0$,有$y\e^{-y}\leqslant\dfrac1\e$.于是对任意$t\geqslant0$和$x\geqslant1$,令$y=tx$,得到
    \[\left|t\e^{-tx}\dfrac{\cos x}{x}\right|
    \leqslant\dfrac{tx\e^{-tx}}{x^2}\leqslant\dfrac{1}{\e x^2}\]
    而$\displaystyle\int_1^{+\infty}\dfrac{1}{\e x^2}\di x=\dfrac1\e<+\infty$,因此根据Weierstrass判别法,原积分在$t\in[0,+\infty)$上一致收敛,且绝对值积分也一致收敛.
    事实上,对任意$A\geqslant1$均有一致的余项估计
    \[\sup_{t\geqslant0}\int_A^{+\infty}\left|t\e^{-tx}\dfrac{\cos x}{x}\right|\di x
    \leqslant\int_A^{+\infty}\dfrac{1}{\e x^2}\di x=\dfrac{1}{\e A}\to0\qquad(A\to+\infty)\]
\end{solution}
\begin{problem}[7.(20\songti{分})]
    设$f(x)$是以$2\pi$为周期的函数,满足$f(x)=|x|,x\in[-\pi,\pi]$.求出$f(x)$的傅里叶级数及其和函数,并求出级数
    \[\sum_{n=1}^{\infty}\dfrac{1}{n^4}\]
    的和.

\end{problem}
\begin{solution}
    注意到$f(x)$为偶函数,因此$b_n=0$,只需计算$a_n$.
    \[a_0=\dfrac1\pi\int_{-\pi}^{\pi}|x|\di x=\pi\]
    \[\begin{aligned}
        a_n
        &=\dfrac1\pi\int_{-\pi}^{\pi}|x|\cos(nx)\di x
        =\dfrac2\pi\int_0^\pi x\cos(nx)\di x \\
        &=\dfrac2\pi\left.\left(\dfrac{x\sin(nx)}{n}+\dfrac{\cos(nx)}{n^2}\right)\right|_0^\pi \\
        &=\dfrac{2}{\pi n^2}\left((-1)^n-1\right)
        =\begin{cases}
            0,&n\text{为偶数},\\
            -\dfrac{4}{\pi n^2},&n\text{为奇数}.
        \end{cases}
    \end{aligned}\]
    于是$f(x)$的Fourier级数为
    \[f(x)\sim\dfrac\pi2-\sum_{n=0}^{\infty}\dfrac{4\cos((2n+1)x)}{\pi(2n+1)^2}\]
    由于$f(x)$的周期延拓在$\R$上连续,且在一个周期内分段光滑,因此根据Dirichlet定理,上述Fourier级数在每个$x\in\R$处都收敛于$f(x)$.其和函数为
    \[S(x)=f(x)=|x-2k\pi|,\qquad
    x\in[(2k-1)\pi,(2k+1)\pi],\quad k\in\mathbb{Z}\]
    根据Parseval等式有
    \[\dfrac{\pi^2}{2}+\sum_{n=0}^{\infty}\dfrac{16}{\pi^2(2n+1)^4}=\dfrac1\pi\int_{-\pi}^{\pi}x^2\di x=\dfrac23\pi^2\]
    于是
    \[\sum_{n=0}^{\infty}\dfrac{1}{(2n+1)^4}=\dfrac{\pi^4}{96}\]
    记$T=\displaystyle\sum_{n=1}^{\infty}\dfrac{1}{n^4}$.按奇偶项拆分,有
    \[T=\sum_{n=0}^{\infty}\dfrac{1}{(2n+1)^4}+\sum_{n=1}^{\infty}\dfrac{1}{(2n)^4}
    =\dfrac{\pi^4}{96}+\dfrac{T}{16}\]
    于是
    \[\sum_{n=1}^{\infty}\dfrac{1}{n^4}=T=\dfrac{16}{15}\cdot\dfrac{\pi^4}{96}=\dfrac{\pi^4}{90}\]

\end{solution}
