\begin{problem}[1.(10\songti{分})]
对于$n\in\N^*$,设
        \[u_{2n-1}=\dfrac1n\ \ \ \ \ u_{2n}=\int_n^{n+1}\dfrac{\di x}{x}\]
        判断级数
        \[\sum_{n=1}^{\infty}(-1)^{n-1}u_n\]
        的敛散性.
\end{problem}
\begin{solution}
记部分和为$S_N$.对偶数部分和配对,有
\[S_{2N}=\sum_{n=1}^N\left(\frac1n-\ln(1+1/n)\right)\]
由$x-x^2/2<\ln(1+x)<x$ $(x>0)$,有
\[0<\frac1n-\ln(1+1/n)<\frac1{2n^2}\]
故$S_{2N}$收敛.又$S_{2N+1}-S_{2N}=1/(N+1)\to0$,所以整个原级数收敛.
其绝对值级数至少包含发散的奇数项子级数$\sum1/n$,故原级数为条件收敛.配对后收敛本身不足以直接推出未配对级数收敛,上述奇数部分和检查不可省略.
\end{solution}

\begin{problem}[2.(10\songti{分})]
求积分
        \[\int_0^{+\infty}\dfrac{\e^{-\alpha x}-\e^{-\beta x}}{x}\sin mx\di x\]
        其中$\alpha>0,\beta>0,m\neq0$.
\end{problem}
\begin{solution}
由$\e^{-\alpha x}-\e^{-\beta x}=x\int_\alpha^\beta\e^{-yx}\di y$,原积分可写为
\[\int_0^{+\infty}\int_\alpha^\beta\e^{-yx}\sin(mx)\di y\di x\]
令$c=\min\{\alpha,\beta\}>0$.相应二重积分的绝对值积分不超过
\[|\beta-\alpha|\int_0^{+\infty}\e^{-cx}\di x<+\infty\]
故可交换积分次序,其中$\alpha>\beta$时按有向积分理解.由此
\[\begin{aligned}
\int_0^{+\infty}\frac{\e^{-\alpha x}-\e^{-\beta x}}x\sin(mx)\di x
&=\int_\alpha^\beta\left(\int_0^{+\infty}\e^{-yx}\sin(mx)\di x\right)\di y\\
&=\int_\alpha^\beta\frac{m}{y^2+m^2}\di y
=\arctan\frac\beta m-\arctan\frac\alpha m
\end{aligned}\]
对$m>0$和$m<0$均成立.
\end{solution}

\begin{problem}[3.(10\songti{分})]
设
        \[\sum_{n=1}^{\infty}a_n\]
        为正项级数,并有
        \[\lim_{n\to\infty}\dfrac{\ln\dfrac{1}{a_n}}{\ln n}=b\]
        \begin{enumerate}[label=\tbf{(\arabic*)},topsep=0pt,parsep=0pt,itemsep=0pt,partopsep=0pt]
            \item \textbf{(5\songti{分})}\ 试证明:当$b>1$时,级数
                \[\sum_{n=1}^{\infty}a_n\]
                收敛.
            \item \textbf{(5\songti{分})}\ 试求$b$的取值范围,使得上述级数一定发散.
        \end{enumerate}
\end{problem}
\begin{solution}
\begin{enumerate}[label=\tbf{(\arabic*)},topsep=0pt,parsep=0pt,itemsep=0pt,partopsep=0pt]
\item 取$1<c<b$.由极限条件,充分大的$n$满足$\ln(1/a_n)/\ln n>c$,故$0<a_n<n^{-c}$.由比较判别法,原级数收敛.
\item 当$b<1$时,取$b<c<1$,则充分大的$n$满足$\ln(1/a_n)/\ln n<c$,所以$a_n>n^{-c}$.由于$\sum n^{-c}$发散,原级数必发散.
当$b=1$时敛散性不确定:取$a_n=1/n$时发散;取$a_n=1/(n\ln^2n)$ $(n\geqslant2$,首项任取正数$)$时收敛,但二者都满足题设极限等于$1$.
结合第(1)问,保证发散的范围恰为$b<1$.
\end{enumerate}
注意对数之比趋于$1$并不能推出原数列之比趋于$1$,不能以此误用极限比较判别法.
\end{solution}

\begin{problem}[4.(10\songti{分})]
求下列函数项级数的收敛区间和收敛域.
        \begin{enumerate}[label=\tbf{(\arabic*)},topsep=0pt,parsep=0pt,itemsep=0pt,partopsep=0pt]
            \item \textbf{(5\songti{分})}\ 
                \[\sum_{n=2}^{\infty}\dfrac{x^{n-1}}{n\cdot3^n\cdot\ln n}\]
            \item \textbf{(5\songti{分})}\ 
                \[\sum_{n=1}^{\infty}\left(\sum_{k=1}^{n}\dfrac{1}{k}\right)x^n\]

        \end{enumerate}
\end{problem}
\begin{solution}
\begin{enumerate}[label=\tbf{(\arabic*)},topsep=0pt,parsep=0pt,itemsep=0pt,partopsep=0pt]
\item 系数相邻比值趋于$1/3$,故收敛半径为$3$,收敛区间为$(-3,3)$.
当$x=3$时,级数为$\frac13\sum_{n=2}^{\infty}1/(n\ln n)$,由积分判别法发散.
当$x=-3$时,级数为$\frac13\sum_{n=2}^{\infty}(-1)^{n-1}/(n\ln n)$,其正项部分单调递减趋于$0$,故由Leibniz判别法收敛.因此收敛域为$[-3,3)$.
\item 记$H_n=\sum_{k=1}^n1/k$,有
\[\frac{H_{n+1}}{H_n}=1+\frac1{(n+1)H_n}\to1\]
故收敛半径为$1$,收敛区间为$(-1,1)$.在$x=\pm1$时,通项$H_n(\pm1)^n$均不趋于$0$,因此均发散,收敛域为$(-1,1)$.
\end{enumerate}
\end{solution}

\begin{problem}[5.(10\songti{分})]
讨论数项级数
        \[\sum_{n=1}^{\infty}\dfrac{\sin n\varphi}{n^p}\]
        的敛散性,其中$\varphi\in(0,\pi)$为取定的参数.
\end{problem}
\begin{solution}
\begin{enumerate}[label=\tbf{\roman*.},topsep=0pt,parsep=0pt,itemsep=0pt,partopsep=0pt,leftmargin=*]
\item 若$p>1$,由$|\sin(n\varphi)|/n^p\leqslant1/n^p$,原级数绝对收敛.
\item 若$0<p\leqslant1$,由有限和公式
\[\sum_{n=1}^N\sin(n\varphi)=\frac{\sin(N\varphi/2)\sin((N+1)\varphi/2)}{\sin(\varphi/2)}\]
可知部分和有界.又$n^{-p}$递减趋于零,故由Dirichlet判别法原级数收敛.
另一方面,对有限部分和有
\[\sum_{n=1}^N\frac{|\sin(n\varphi)|}{n^p}
\geqslant\frac12\sum_{n=1}^N\frac1{n^p}
-\frac12\sum_{n=1}^N\frac{\cos(2n\varphi)}{n^p}\]
由于$0<\varphi<\pi$,余弦部分和有界,故右端第二级数收敛,第一级数发散.因此原级数条件收敛.
\item 若$p\leqslant0$,假设通项趋于零,则$\sin(n\varphi)\to0$.由
\[\sin((n+1)\varphi)-\sin((n-1)\varphi)=2\cos(n\varphi)\sin\varphi\]
及$\sin\varphi\neq0$,得$\cos(n\varphi)\to0$,与$\sin^2(n\varphi)+\cos^2(n\varphi)=1$矛盾.故原级数发散.
\end{enumerate}
\end{solution}

\begin{problem}[6.(10\songti{分})]
判断下列广义积分的敛散性.
        \begin{enumerate}[label=\tbf{(\arabic*)},topsep=0pt,parsep=0pt,itemsep=0pt,partopsep=0pt]
            \item \textbf{(5\songti{分})}
                \[\int_0^{+\infty}\dfrac{\ln(1+x)}{x^p}\di x\]
            \item \textbf{(5\songti{分})}
                \[\int_0^{\frac12}\dfrac{\ln x}{\sqrt{x}\left(1-x\right)^2}\di x\]

        \end{enumerate}
\end{problem}
\begin{solution}
\begin{enumerate}[label=\tbf{(\arabic*)},topsep=0pt,parsep=0pt,itemsep=0pt,partopsep=0pt]
\item 当$x\to0^+$时,$\ln(1+x)/x^p\sim x^{1-p}$,所以零点处收敛当且仅当$p<2$.
在无穷远处,若$p>1$,取$q=(p+1)/2>1$,则$\ln(1+x)/x^p=o(x^{-q})$,故收敛.
若$p\leqslant1$,则$x\geqslant1$时$\ln(1+x)/x^p\geqslant\ln x/x$,而$\int_1^{+\infty}\ln x/x\di x$发散.
故原积分当且仅当$1<p<2$时收敛,其余情形发散于$+\infty$.
\item 在$0<x\leqslant1/2$上,
\[\left|\frac{\ln x}{\sqrt x(1-x)^2}\right|\leqslant\frac{4(-\ln x)}{\sqrt x}\]
又作$x=u^2$代换得
\[\int_0^{1/2}\frac{-\ln x}{\sqrt x}\di x
=-4\int_0^{1/\sqrt2}\ln u\di u=\sqrt2(2+\ln2)<+\infty\]
因此原瑕积分绝对收敛.
\end{enumerate}
\end{solution}

\begin{problem}[7.(10\songti{分})]
求含参变量$x$的无穷积分
        \[I(x)=\int_0^{+\infty}\e^{-t^2}\cos 2xt\di t\]
\end{problem}
\begin{solution}
原被积函数由$\e^{-t^2}$控制,其对参数$x$的偏导数为$-2t\e^{-t^2}\sin(2xt)$,绝对值由可积函数$2t\e^{-t^2}$控制.因此可以在积分号下求导,并分部积分得
\[\begin{aligned}
I'(x)&=\int_0^{+\infty}\sin(2xt)\di(\e^{-t^2})\\
&=\left.\e^{-t^2}\sin(2xt)\right|_0^{+\infty}
-2x\int_0^{+\infty}\e^{-t^2}\cos(2xt)\di t=-2xI(x)
\end{aligned}\]
由$I(0)=\sqrt\pi/2$,解得
\[I(x)=\frac{\sqrt\pi}{2}\e^{-x^2},\qquad x\in\R\]
\end{solution}

\begin{problem}[8.(15\songti{分})]
设$f(x)$以$2\pi$为周期,且在$[-\pi,\pi]$上可积的函数.$a_n,b_n$为$f(x)$的Fourier系数.
        \begin{enumerate}[label=\tbf{(\arabic*)},topsep=0pt,parsep=0pt,itemsep=0pt,partopsep=0pt]
            \item \textbf{(3\songti{分})}\ 试求延迟函数$f(x+t)$的Fourier系数.
            \item \textbf{(12\songti{分})}\ 设$f(x)$连续且在$[-\pi,\pi]$上分段光滑,试求卷积函数
                \[F(x)=\dfrac{1}{\pi}\int_{-\pi}^{\pi}f(t)f(x+t)\di t\]
                的Fourier展开式,并由此推出Parseval等式.

        \end{enumerate}
\end{problem}
\begin{solution}
\begin{enumerate}[label=\tbf{(\arabic*)},topsep=0pt,parsep=0pt,itemsep=0pt,partopsep=0pt]
\item 由换元$u=x+t$及周期性可得,常数项系数仍为$a_0$,且$n\geqslant1$时
\[\begin{aligned}
a_n'&=\frac1\pi\int_{-\pi}^\pi f(x+t)\cos(nx)\di x
=a_n\cos(nt)+b_n\sin(nt),\\
b_n'&=\frac1\pi\int_{-\pi}^\pi f(x+t)\sin(nx)\di x
=b_n\cos(nt)-a_n\sin(nt).
\end{aligned}\]
\item 被积函数在有限矩形上连续,可交换积分次序.设$F$的系数为$A_n,B_n$,则
\[A_0=\frac1\pi\int_{-\pi}^\pi F(x)\di x=a_0^2\]
对$n\geqslant1$,利用第(1)问得到
\[\begin{aligned}
A_n&=\frac1\pi\int_{-\pi}^\pi F(x)\cos(nx)\di x\\
&=\frac1\pi\int_{-\pi}^\pi f(t)\bigl(a_n\cos(nt)+b_n\sin(nt)\bigr)\di t
=a_n^2+b_n^2,\\
B_n&=\frac1\pi\int_{-\pi}^\pi f(t)\bigl(b_n\cos(nt)-a_n\sin(nt)\bigr)\di t=0.
\end{aligned}\]
由于$f$连续且分段光滑,在有限段上分部积分可得
\[F'(x)=-\frac1\pi\int_{-\pi}^\pi f'(t)f(x+t)\di t\]
右端对$x$连续,故$F$连续可导且以$2\pi$为周期.由Dirichlet定理,
\[F(x)=\frac{a_0^2}{2}+\sum_{n=1}^{\infty}(a_n^2+b_n^2)\cos(nx)\]
令$x=0$,得到
\[\frac1\pi\int_{-\pi}^\pi f(t)^2\di t
=F(0)=\frac{a_0^2}{2}+\sum_{n=1}^{\infty}(a_n^2+b_n^2)\]
即Parseval等式.
\end{enumerate}
\end{solution}

\begin{problem}[9.(15\songti{分})]
回答下列问题.
        \begin{enumerate}[label=\tbf{(\arabic*)},topsep=0pt,parsep=0pt,itemsep=0pt,partopsep=0pt]
            \item \textbf{(5\songti{分})}\ 把$f(x)=x^2$在$(-\pi,\pi]$上展开为Fourier级数.
            \item \textbf{(5\songti{分})}\ 利用\tbf{(1)}的结论证明
                \[\sum_{n=1}^{\infty}\dfrac{1}{n^2}=\dfrac{\pi^2}{6}\ \ \ \ \ \sum_{n=1}^{\infty}\dfrac{1}{(2n-1)^2}=\dfrac{\pi^2}{8}\]
            \item \textbf{(5\songti{分})}\ 利用Parseval等式求级数
                \[\sum_{n=1}^{\infty}\dfrac{1}{n^4}\]
                的值.
        \end{enumerate}
\end{problem}
\begin{solution}
\begin{enumerate}[label=\tbf{(\arabic*)},topsep=0pt,parsep=0pt,itemsep=0pt,partopsep=0pt]
\item 偶性给出$b_n=0$,直接分部积分得$a_0=2\pi^2/3$, $a_n=4(-1)^n/n^2$.周期延拓连续且分段光滑,因此在$(-\pi,\pi]$上
\[x^2=\frac{\pi^2}{3}+4\sum_{n=1}^{\infty}\frac{(-1)^n\cos(nx)}{n^2}\]
\item 令$x=\pi$,得到$\sum1/n^2=\pi^2/6$.按奇偶项拆分,得到
\[\sum_{n=1}^{\infty}\frac1{(2n-1)^2}
=\left(1-\frac14\right)\sum_{n=1}^{\infty}\frac1{n^2}=\frac{\pi^2}{8}\]
\item 由Parseval等式,
\[\frac{2\pi^4}{9}+16\sum_{n=1}^{\infty}\frac1{n^4}
=\frac1\pi\int_{-\pi}^\pi x^4\di x=\frac{2\pi^4}{5}\]
因此$\displaystyle\sum_{n=1}^{\infty}1/n^4=\pi^4/90$.
\end{enumerate}
\end{solution}
