\begin{problem}[1.(15\songti{分})]
判断下列级数的敛散性.
        \begin{enumerate}[label=\tbf{(\arabic*)},topsep=0pt,parsep=0pt,itemsep=0pt,partopsep=0pt]
            \item \textbf{(5\songti{分})}
                \[\sum_{n=3}^{\infty}\dfrac{2}{n\ln\sqrt{n}}\]
            \item \textbf{(5\songti{分})}
                \[\sum_{n=1}^{\infty}\dfrac{3\sqrt[5]{n}+1}{\left(\sqrt[4]{n}+n\right)\left(\sqrt[3]{n}+n\right)}\]
            \item \textbf{(5\songti{分})}
                \[\sum_{n=1}^{\infty}\left(\dfrac{4\cdot3^n}{5^n}-\dfrac{1}{n}+\dfrac{1}{n+2}\right)\]
        \end{enumerate}
\end{problem}
\begin{solution}
\begin{enumerate}[label=\tbf{(\arabic*)},topsep=0pt,parsep=0pt,itemsep=0pt,partopsep=0pt]
\item 因为$2/(n\ln\sqrt n)=4/(n\ln n)$,且后者随$n$递减,所以
\[\sum_{n=3}^N\frac4{n\ln n}\geqslant\int_3^{N+1}\frac4{x\ln x}\di x
=4\bigl(\ln\ln(N+1)-\ln\ln3\bigr)\to+\infty\]
原级数发散.
\item 对$n\geqslant1$有
\[0<\frac{3\sqrt[5]n+1}{(\sqrt[4]n+n)(\sqrt[3]n+n)}
\leqslant\frac{3n^{1/5}+1}{n^2}\leqslant\frac4{n^{9/5}}\]
由比较判别法,原级数收敛.
\item 几何级数$\sum_{n=1}^{\infty}4(3/5)^n=6$,而
\[\sum_{n=1}^N\left(\frac1n-\frac1{n+2}\right)
=\frac32-\frac1{N+1}-\frac1{N+2}\to\frac32\]
故原级数收敛,其和为$6-3/2=9/2$.
\end{enumerate}
\end{solution}

\begin{problem}[2.(10\songti{分})]
讨论函数序列
        \[f_n(x)=\left(1-\dfrac{1}{\sqrt{n}}\right)^{x^2}\ \ \ \ \ n=1,2,\cdots\]
        在$(0,+\infty)$上的一致收敛性.
\end{problem}
\begin{solution}
对每个固定$x>0$,有$f_n(x)\to1$.但取$x_n=n^{1/4}$,则
\[|f_n(x_n)-1|=1-\left(1-\frac1{\sqrt n}\right)^{\sqrt n}\to1-\e^{-1}>0\]
故不满足一致收敛的必要条件,原函数序列在$(0,+\infty)$上不一致收敛.
事实上,对每个$n$,当$x\to+\infty$时$f_n(x)\to0$,所以$\sup_{x>0}|f_n(x)-1|=1$.
\end{solution}

\begin{problem}[3.(15\songti{分})]
求幂级数
        \[\sum_{n=1}^{\infty}\dfrac{(-1)^n}{n}x^n\]
        的收敛半径,收敛域,和函数.
\end{problem}
\begin{solution}
由系数比值$n/(n+1)\to1$可知收敛半径为$R=1$.当$x=1$时是收敛的交错调和级数,但不绝对收敛;当$x=-1$时是发散的调和级数.故收敛域为$(-1,1]$.
在$|x|<1$时逐项求导,得到
\[S'(x)=\sum_{n=1}^{\infty}(-1)^nx^{n-1}=-\frac1{1+x}\]
由$S(0)=0$,得到$S(x)=-\ln(1+x)$.由Abel定理,该公式在收敛端点$x=1$也成立,因此
\[S(x)=-\ln(1+x),\qquad -1<x\leqslant1\]
\end{solution}

\begin{problem}[4.(10\songti{分})]
求函数
        \[f(x)=\dfrac{1}{x^2-2x-3}\]
        于$x=1$处的泰勒展开式,并计算$f^{(2022)}(1),f^{(2023)}(1)$的值.
\end{problem}
\begin{solution}
有
\[f(x)=-\frac14\frac1{1-(x-1)^2/4}
=-\sum_{n=0}^{\infty}\frac{(x-1)^{2n}}{4^{n+1}},\qquad |x-1|<2\]
比较Taylor系数,得到
\[f^{(2022)}(1)=-\frac{2022!}{4^{1012}},\qquad f^{(2023)}(1)=0\]
\end{solution}

\begin{problem}[5.(10\songti{分})]
讨论无穷积分
        \[\int_1^{+\infty}\dfrac{\sin x}{\sqrt{x}}\arctan x\di x\]
        的敛散性.
\end{problem}
\begin{solution}
由Dirichlet判别法,$\int_1^{+\infty}\sin x/\sqrt x\di x$收敛.又$\arctan x$单调有界,由Abel判别法可知原积分收敛.
另一方面,对整数$N\geqslant2$,有
\[\begin{aligned}
\int_\pi^{N\pi}\frac{|\sin x|\arctan x}{\sqrt x}\di x
&\geqslant\arctan\pi\sum_{k=1}^{N-1}\frac1{\sqrt{(k+1)\pi}}
\int_{k\pi}^{(k+1)\pi}|\sin x|\di x\\
&=\frac{2\arctan\pi}{\sqrt\pi}\sum_{k=1}^{N-1}\frac1{\sqrt{k+1}}\to+\infty
\end{aligned}\]
因此原积分不绝对收敛,结论为条件收敛.
\end{solution}

\begin{problem}[6.(10\songti{分})]
讨论级数
        \[\sum_{n=1}^{\infty}\dfrac{\sin^n x}{1+\sin^{2n}x}\]
        在$x\in(-\infty,+\infty)$上的绝对收敛性和条件收敛性.
\end{problem}
\begin{solution}
令$u=\sin x$.若$|u|<1$,则
\[\left|\frac{u^n}{1+u^{2n}}\right|\leqslant|u|^n\]
由几何级数比较可知原级数绝对收敛,其中$u=0$时各项均为零.
若$u=1$,通项等于$1/2$;若$u=-1$,通项等于$(-1)^n/2$.两种情形均因通项不趋于零而发散.
故全体绝对收敛点为
\[\R\setminus\{\pi/2+k\pi:k\in\mathbb Z\}\]
不存在条件收敛点,其余点均发散.
\end{solution}

\begin{problem}[7.(20\songti{分})]
设以$2\pi$为周期的函数$f(x)$在$[-\pi,\pi]$上的表达式为
        \[f(x)=x^2\]
        求$f(x)$的Fourier级数及其和函数,并给出级数
        \[\sum_{n=1}^{\infty}\dfrac{1}{n^2}
        \ \ \ \ \ \sum_{n=1}^{\infty}\dfrac{(-1)^n}{n^2}
        \ \ \ \ \ \sum_{n=1}^{\infty}\dfrac{1}{n^4}\]
        的值.
\end{problem}
\begin{solution}
由偶性知$b_n=0$,且
\[a_0=\frac{2\pi^2}{3},\qquad
a_n=\frac2\pi\int_0^\pi x^2\cos(nx)\di x=\frac{4(-1)^n}{n^2}\]
周期延拓在$\R$上连续且分段光滑,由Dirichlet定理,
\[f(x)=\frac{\pi^2}{3}+4\sum_{n=1}^{\infty}\frac{(-1)^n\cos(nx)}{n^2}\]
和函数在每段$[(2k-1)\pi,(2k+1)\pi]$上等于$(x-2k\pi)^2$.
分别令$x=\pi$和$x=0$,得到
\[\sum_{n=1}^{\infty}\frac1{n^2}=\frac{\pi^2}{6},\qquad
\sum_{n=1}^{\infty}\frac{(-1)^n}{n^2}=-\frac{\pi^2}{12}\]
再由Parseval等式,
\[\frac{2\pi^4}{9}+16\sum_{n=1}^{\infty}\frac1{n^4}
=\frac1\pi\int_{-\pi}^\pi x^4\di x=\frac{2\pi^4}{5}\]
故$\displaystyle\sum_{n=1}^{\infty}1/n^4=\pi^4/90$.
\end{solution}

\begin{problem}[8.(10\songti{分})]
证明和计算下列各题.
        \begin{enumerate}[label=\tbf{(\arabic*)},topsep=0pt,parsep=0pt,itemsep=0pt,partopsep=0pt]
            \item 证明含参变量$t$的无穷积分
                \[I(t)=\int_0^{+\infty}\e^{-tx}\dfrac{\sin x}{x}\di x\]
                在$t\in[0,+\infty)$上一致收敛.
            \item 证明上述$I(t)$在$t\in(0,+\infty)$上可导.
            \item 求出上述$I(t)$.
            \item 计算无穷积分
                \[\int_0^{+\infty}\dfrac{\sin x}{x}\di x\]
                的值.
        \end{enumerate}
\end{problem}
\begin{solution}
\begin{enumerate}[label=\tbf{(\arabic*)},topsep=0pt,parsep=0pt,itemsep=0pt,partopsep=0pt]
\item 在$x=0$处把被积函数连续延拓为$1$.对$x\geqslant A\geqslant1$和$t\geqslant0$,令$h_t(x)=\e^{-tx}/x$,则$h_t$非负、单调递减,且$h_t(A)\leqslant1/A$.分部积分得
\[\left|\int_A^B h_t(x)\sin x\di x\right|\leqslant2h_t(A)\leqslant\frac2A\qquad(B>A)\]
由一致Cauchy准则,原积分在$t\in[0,+\infty)$上一致收敛.
\item 在任意$[a,b]\subset(0,+\infty)$上,偏导数为$-\e^{-tx}\sin x$,其绝对值不超过可积函数$\e^{-ax}$.因此可以在积分号下求导,得到
\[I'(t)=-\int_0^{+\infty}\e^{-tx}\sin x\di x=-\frac1{1+t^2}\]
\item 故$I(t)=-\arctan t+C$.又$|\sin x|/x\leqslant1$,所以$t>0$时
\[|I(t)|\leqslant\int_0^{+\infty}\e^{-tx}\di x=\frac1t\to0\]
从而$C=\pi/2$,即$I(t)=\pi/2-\arctan t$ $(t>0)$.
\item 由第(1)问的一致收敛性及被积函数对$t$的连续性,$I(t)$在$t=0$处连续.于是
\[\int_0^{+\infty}\frac{\sin x}{x}\di x=I(0)=\lim_{t\to0^+}I(t)=\frac\pi2\]
\end{enumerate}
\end{solution}
