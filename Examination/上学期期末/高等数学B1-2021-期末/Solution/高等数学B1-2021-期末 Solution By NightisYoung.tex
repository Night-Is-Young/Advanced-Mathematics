\begin{problem}[1.(10\songti{分})]
证明:对于任意$x\in\R$,存在$\theta\in(0,1)$使得
        \[\arctan x=\dfrac{x}{1+\theta^2x^2}\]
        成立.
\end{problem}
\begin{solution}
对$f(u)=\arctan u$在$0$与$x$之间应用Lagrange中值定理.
当$x\neq0$时存在位于二者之间的$\xi$满足$\arctan x=x/(1+\xi^2)$.
令$\theta=\xi/x$,则$0<\theta<1$,得到所需等式.当$x=0$时任取$\theta\in(0,1)$即可.
\end{solution}
\begin{problem}[2.(20\songti{分})]
求出下面函数的极限.
        \begin{enumerate}[label=\textbf{(\arabic*)},leftmargin=*]
            \item \textbf{(10\songti{分})}\ $\displaystyle\lim_{x\to0}\dfrac{\tan^4x}{\sqrt{1-\dfrac{x\sin x}{2}}-\sqrt{\cos x}}$.
            \item \textbf{(10\songti{分})}\ 设$n\in\N^*$.对于实序列$\left\{a_k\right\}_{k=1}^n$,求\[\lim_{x\to0}\left(\dfrac{\sum_{k=1}^na_k^x}{n}\right)^{\frac{1}{x}}\]
        \end{enumerate}
\end{problem}
\begin{solution}
\begin{enumerate}[label=\tbf{(\arabic*)},leftmargin=*]
\item 将分母有理化.由$\sin x=x-x^3/6+o(x^3)$和$\cos x=1-x^2/2+x^4/24+o(x^4)$,
\[1-\dfrac{x\sin x}{2}-\cos x=\dfrac{x^4}{24}+o(x^4)\]
而$\tan^4x=x^4+o(x^4)$,两根式之和趋于$2$,所以原极限为$48$.
\item 现有题干未写$a_k>0$,一般实数$a_k$不能保证$a_k^x$在$x=0$的邻域内有实数意义.
在补充$a_1,\ldots,a_n>0$的条件下,取对数并应用L'Hospital法则,
\[\lim_{x\to0}\dfrac{\ln(\sum_{k=1}^n a_k^x)-\ln n}{x}
=\dfrac1n\sum_{k=1}^n\ln a_k\]
因此原极限为$\sqrt[n]{a_1\cdots a_n}$.
\end{enumerate}
\end{solution}
\begin{problem}[3.(15\songti{分})]
设函数\[f(x)=\dfrac{1-2x+5x^2}{(1-2x)(1+x^2)}\]在$x=0$处的$2n+1$阶泰勒公式.
\end{problem}
\begin{solution}
将有理函数拆成
\[f(x)=-\dfrac{2x}{1+x^2}+\dfrac1{1-2x}\]
在$x=0$附近展开为几何级数并截断,得到
\[f(x)=\sum_{j=0}^{2n+1}2^jx^j-2\sum_{j=0}^n(-1)^jx^{2j+1}+o(x^{2n+1})\]
这就是所求的$2n+1$阶局部泰勒公式;其中$x$的一次项相消.
\end{solution}
\begin{problem}[4.(10\songti{分})]
定义三元函数$f:\R^3\to\R$为\[f(x,y,z)=\left\{\begin{array}{l}
            \dfrac{xyz}{x^2+y^2+z^2},(x,y,z)\neq(0,0,0)\\
            0,(x,y,z)=(0,0,0)
        \end{array}\right.\]回答下列问题.
        \begin{enumerate}[label=\textbf{(\arabic*)},leftmargin=*]
            \item \textbf{(5\songti{分})}\ 求函数$f(x,y,z)$在$(0,0,0)$处的三个偏导数.
            \item \textbf{(5\songti{分})}\ $f(x,y,z)$在$(0,0,0)$处是否可微?试证明之.
        \end{enumerate}
\end{problem}
\begin{solution}
\begin{enumerate}[label=\tbf{(\arabic*)},leftmargin=*]
\item 在任一坐标轴上$f=0$,故在原点的三个偏导数均为$0$.
\item 若在原点可微,由三个偏导数均为零,应有$f(x,y,z)=o(\rho)$,其中$\rho=\sqrt{x^2+y^2+z^2}$.
但沿$x=y=z=t>0$,有
\[\dfrac{f(t,t,t)}{\sqrt{3t^2}}=\dfrac{t/3}{\sqrt3t}=\dfrac1{3\sqrt3}\neq0\]
故$f$在原点不可微.偏导数不连续本身不能作为不可微的充分理由.
\end{enumerate}
\end{solution}
\begin{problem}[5.(15\songti{分})]
设$f,g:\R\to\R$都有连续的二阶导数.对于任意$x,y\in\R$,
        定义$h(x,y)=xg\left(\dfrac{y}{x}\right)+f\left(\dfrac{y}{x}\right)$,
        试计算$x^2h_{xx}(x,y)+2xyh_{yx}(x,y)+y^2h_{yy}(x,y)$.
\end{problem}
\begin{solution}
公式在$x\neq0$的区域内有定义.固定$(x,y)$,令$H(s)=h(sx,sy)$,其中$s$在$1$附近,则
\[H(s)=sxg(y/x)+f(y/x)\]
故$H''(1)=0$.另一方面,链式法则给出
\[H''(1)=x^2h_{xx}+2xyh_{yx}+y^2h_{yy}\]
所以所求结果为$0$.现有题干所写的任意$x,y$应理解为$x\neq0$的定义域内.
\end{solution}
\begin{problem}[6.(20\songti{分})]
设函数$F:\R^3\to\R$为\[F(x,y,z)=x^3+(y^2-1)z^3-xyz\]回答下列问题.
        \begin{enumerate}[label=\textbf{(\arabic*)},leftmargin=*]
            \item \textbf{(5\songti{分})}\ 证明:存在$\R^2$上$(1,1)$的邻域$D$使得$D$上由$F(x,y,z)\equiv0$确定唯一隐函数$z=f(x,y)$且$f(1,1)=1$.
            \item \textbf{(5\songti{分})}\ 求出在$(1,1)$处函数$z=f(x,y)$减少最快的方向上的单位向量$\vec{v}$.
            \item \textbf{(10\songti{分})}\ 设$\R^3$中平面$x+2y-2z=1$的$z$分量为正的法向量记为$\vec{u}$.向量$(\vec{v},0)\in\R^3$.求$\vec{u}$与$(\vec{v},0)$的夹角余弦.
        \end{enumerate}
\end{problem}
\begin{solution}
\begin{enumerate}[label=\tbf{(\arabic*)},leftmargin=*]
\item $F(1,1,1)=0$,且$F_z(1,1,1)=-1\neq0$.由隐函数定理,在$(1,1)$附近存在唯一光滑的局部隐函数$z=f(x,y)$,满足$f(1,1)=1$.
\item 在$(1,1,1)$处,$F_x=2,F_y=1,F_z=-1$,故$\nabla f(1,1)=(2,1)$.
下降最快的单位方向为$\vec v=(-2,-1)/\sqrt5$.
\item 可取$z$分量为正的法向量$\vec u=(-1,-2,2)$,故
\[\cos\langle\vec u,(\vec v,0)\rangle
=\dfrac{(-1,-2,2)\cdot(-2,-1,0)}{3\sqrt5}=\dfrac4{3\sqrt5}=\dfrac{4\sqrt5}{15}\]
\end{enumerate}
\end{solution}
\begin{problem}[7.(10\songti{分})]
给定正整数$n\geqslant 3$,求单位圆的内接$n$边形面积的最大值.
\end{problem}
\begin{solution}
按圆周顺序排列各顶点,设相邻顶点之间的有向圆心角为$\theta_1,\ldots,\theta_n$,其中$\theta_i\geqslant0$,且$\sum\theta_i=2\pi$.
由行列式面积公式,多边形面积为$S=\frac12\sum\sin\theta_i$.
在这个闭单纯形上最大值存在.若某个$\theta_i=0$,则可将另一个$0<\theta_j<2\pi$分成两个相等的角,使面积增加,因为$2\sin(\theta_j/2)>\sin\theta_j$.
全体重合的退化情形面积为$0$,也不是最大值,故最大值点在内部.
由Lagrange乘子法,所有$\cos\theta_i$相等.若有一个角大于$\pi$,设它为$2\pi-\alpha$,其中$0<\alpha<\pi$,则其余角只能是$\alpha$;这将使总和等于$2\pi+(n-2)\alpha>2\pi$,矛盾.
因此所有角都小于$\pi$,进而全部相等,即$\theta_i=2\pi/n$.
所以最大面积为
\[S_{\max}=\dfrac n2\sin\dfrac{2\pi}{n}\]
等号当且仅当多边形为正$n$边形.
\end{solution}
